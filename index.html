Com o que este protótipo está relacionado?

Trend de parar o cronômetro no tempo solicitado: uma pessoa escolhe um tempo-alvo e o jogador tenta parar o cronômetro o mais próximo possível desse valor.

Imagens do protótipo

Lista de Componentes

  • Placa Arduino Uno 1 un.
  • Display TM1637 (4 Dígitos) 1 un.
  • Buzzer Passivo 1 un.
  • Botão (Push Button) 1 un.
  • Protoboard1 un.
  • Jumpers Vários

Esquema de Pinagem

  • Display TM1637 - CLK Pino 2
  • Display TM1637 - DIO Pino 3
  • Push Button Pino 4
  • Buzzer Passivo Pino 5
Código C++ (Arduino) 
#include <TM1637Display.h>

const int CLK = 2;
const int DIO = 3;
const int PIN_BOTAO = 4;
const int PIN_BUZZER = 5;

TM1637Display display(CLK, DIO);

unsigned long tempoAnterior = 0;
unsigned long tempoDecorrido = 0;
bool cronometroAtivo = false;

bool botaoPressionadoAnteriormente = false;
unsigned long tempoPressionado = 0;
unsigned long tempoUltimoClique = 0;

const int debounceDelay = 50;      
const int tempoCliqueLongo = 1500;  

void setup() {
  display.setBrightness(0x0f);
  pinMode(PIN_BOTAO, INPUT_PULLUP);
  pinMode(PIN_BUZZER, OUTPUT);
  exibirTempo(0);
}

void loop() {
  int estadoBotao = digitalRead(PIN_BOTAO);
  unsigned long tempoAtual = millis();

  if (estadoBotao == LOW && !botaoPressionadoAnteriormente) {
    tempoPressionado = tempoAtual;
    botaoPressionadoAnteriormente = true;
  }

  if (estadoBotao == LOW && botaoPressionadoAnteriormente) {
    if (tempoAtual - tempoPressionado >= tempoCliqueLongo) {
      tone(PIN_BUZZER, 300, 150);
      delay(150);
      tone(PIN_BUZZER, 150, 250);

      cronometroAtivo = false;
      tempoDecorrido = 0;
      exibirTempo(0);
     
      display.setBrightness(0x00);
      delay(100);
      display.setBrightness(0x0f);
      exibirTempo(0);

      while(digitalRead(PIN_BOTAO) == LOW) { delay(10); }
      botaoPressionadoAnteriormente = false;
    }
  }

  if (estadoBotao == HIGH && botaoPressionadoAnteriormente) {
    unsigned long duracaoClique = tempoAtual - tempoPressionado;

    if (duracaoClique >= debounceDelay && duracaoClique < tempoCliqueLongo) {
      cronometroAtivo = !cronometroAtivo;
     
      if (cronometroAtivo) {
        tone(PIN_BUZZER, 2000, 80);
        delay(100);
        tone(PIN_BUZZER, 2500, 80);
       
        tempoAnterior = millis() - tempoDecorrido;
      } else {
        tone(PIN_BUZZER, 1000, 200);
      }
    }
    botaoPressionadoAnteriormente = false;
  }

  if (cronometroAtivo) {
    tempoDecorrido = millis() - tempoAnterior;
    exibirTempo(tempoDecorrido);
  }
}

void exibirTempo(unsigned long ms) {
  unsigned long totalCenteseconds = ms / 10;
  int segundos = (totalCenteseconds / 100) % 100;
  int centesimos = totalCenteseconds % 100;
  int valorDisplay = (segundos * 100) + centesimos;

  display.showNumberDecEx(valorDisplay, 0b01000000, true);
}

Com o que este protótipo está relacionado?

Desafio de contar as batidas: o sensor detecta palmas ou batidas em um intervalo de tempo, e o protótipo apresenta o resultado por meio dos LEDs e do movimento do servo motor.

Imagens do protótipo

Lista de Componentes

  • Placa Arduino Uno 1 un.
  • Sensor de Som Digital (KY-037/KY-038) 1 un.
  • Servo Motor 1 un.
  • LEDs Difusos 7 un.
  • Resistores de 220Ω 7 un.

Esquema de Pinagem

  • Sensor de Som Digital Pino 2
  • LEDs 1 a 7 Pinos 3 a 9
  • Sinal do Servo Motor Pino 10
Código C++ (Arduino) 
#include <Servo.h>

const int pinoSom = 2;          
const int pinosLeds[7] = {3, 4, 5, 6, 7, 8, 9}; 
const int pinoServo = 10;       

Servo meuServo;

volatile int contadorBatidas = 0;
volatile unsigned long tempoUltimaBatida = 0;
volatile unsigned long tempoPrimeiraBatida = 0;
volatile bool novaBatidaDetectada = false; 

const unsigned long janelaTempo = 1500; 
const unsigned long debounceTime = 180; 

void setup() {
  Serial.begin(9600);
 
  for (int i = 0; i < 7; i++) {
    pinMode(pinosLeds[i], OUTPUT);
    digitalWrite(pinosLeds[i], LOW); 
  }
 
  pinMode(pinoSom, INPUT);
  attachInterrupt(digitalPinToInterrupt(pinoSom), somDetectado, FALLING);
 
  meuServo.attach(pinoServo);
  meuServo.write(0); 
 
  Serial.println("=========================================");
  Serial.println("Sistema Pronto! Aguardando batidas...");
  Serial.println("=========================================");
}

void loop() {
  if (novaBatidaDetectada) {
    novaBatidaDetectada = false;
    Serial.print("Batida detectada! Contagem atual: ");
    Serial.println(contadorBatidas);
  }

  if (contadorBatidas > 0 && (millis() - tempoPrimeiraBatida > janelaTempo)) {
    detachInterrupt(digitalPinToInterrupt(pinoSom));
   
    int batidasFinais = contadorBatidas;
    if (batidasFinais > 7) batidasFinais = 7; 
   
    Serial.println("\n--- FIM DA JANELA DE TEMPO ---");
    Serial.print("Total final de batidas: ");
    Serial.println(contadorBatidas);

    for (int i = 0; i < 7; i++) {
      digitalWrite(pinosLeds[i], LOW);
    }

    int angulo = batidasFinais * 20;
    meuServo.write(angulo);
   
    int indiceLed = batidasFinais - 1;
    for (int p = 0; p < 3; p++) {
      digitalWrite(pinosLeds[indiceLed], HIGH);
      delay(200);
      digitalWrite(pinosLeds[indiceLed], LOW);
      delay(200);
    }
   
    digitalWrite(pinosLeds[indiceLed], HIGH);
    contadorBatidas = 0;
    attachInterrupt(digitalPinToInterrupt(pinoSom), somDetectado, FALLING);
  }
}

void somDetectado() {
  unsigned long tempoAtual = millis();
 
  if (tempoAtual - tempoUltimaBatida > debounceTime) {
    if (contadorBatidas == 0) {
      tempoPrimeiraBatida = tempoAtual; 
    }
    contadorBatidas++;
    tempoUltimaBatida = tempoAtual;
    novaBatidaDetectada = true; 
  }
}

Com o que este protótipo está relacionado?

Jogo da memória: o jogador observa uma sequência de luzes e sons e precisa repeti-la pressionando os botões correspondentes, exercitando a atenção e a memória.

Imagens do protótipo

Lista de Componentes

  • Arduino Uno 1 un.
  • Botões (push buttons) 4 un.
  • LEDs 4 un.
  • Buzzer 1 un.
  • Protoboard1 un.
  • Jumpers Vários
  • Resistores para os LEDs 4 un.

Esquema de Pinagem

  • Botão 1, 2, 3 e 4 Pino 1, 2, 3 e 4
  • LEDs 1, 2, 3 e 4 Pinos 8, 9, 10 e 11
  • Buzzer Pino 7
Código C++ (Arduino) 
const int pinosBotoes[4] = {1, 2, 3, 4};
const int pinosLeds[4] = {8, 9, 10, 11};
const int pinoBuzzer = 7;

const int TAM_MAX = 50;
int sequencia[TAM_MAX];
int nivel = 1;

void tocarNota(int frequencia, int duracao) {
  tone(pinoBuzzer, frequencia, duracao);
  delay(duracao);
  noTone(pinoBuzzer);
}

void piscarLed(int indice, int duracao) {
  digitalWrite(pinosLeds[indice], HIGH);
  tocarNota(440 + indice * 110, duracao);
  digitalWrite(pinosLeds[indice], LOW);
  delay(150);
}

void mostrarSequencia() {
  delay(500);
  for (int i = 0; i < nivel; i++) {
    piscarLed(sequencia[i], 350);
  }
}

int lerBotao() {
  for (int i = 0; i < 4; i++) {
    if (digitalRead(pinosBotoes[i]) == LOW) {
      delay(25); // debounce simples
      if (digitalRead(pinosBotoes[i]) == LOW) {
        while (digitalRead(pinosBotoes[i]) == LOW) {
          delay(10);
        }
        return i;
      }
    }
  }
  return -1;
}

void indicarErro() {
  for (int repeticao = 0; repeticao < 3; repeticao++) {
    for (int i = 0; i < 4; i++) {
      digitalWrite(pinosLeds[i], HIGH);
    }
    tocarNota(180, 250);
    for (int i = 0; i < 4; i++) {
      digitalWrite(pinosLeds[i], LOW);
    }
    delay(180);
  }
}

void setup() {
  for (int i = 0; i < 4; i++) {
    pinMode(pinosBotoes[i], INPUT_PULLUP);
    pinMode(pinosLeds[i], OUTPUT);
    digitalWrite(pinosLeds[i], LOW);
  }
  pinMode(pinoBuzzer, OUTPUT);

  randomSeed(analogRead(A0));
  for (int i = 0; i < TAM_MAX; i++) {
    sequencia[i] = random(0, 4);
  }

  mostrarSequencia();
}

void loop() {
  for (int i = 0; i < nivel; i++) {
    int botao = -1;
    while (botao == -1) {
      botao = lerBotao();
    }

    piscarLed(botao, 180);

    if (botao != sequencia[i]) {
      indicarErro();
      nivel = 1;
      mostrarSequencia();
      return;
    }
  }

  tocarNota(1200, 120); // acerto
  delay(400);

  if (nivel < TAM_MAX) {
    nivel++;
    mostrarSequencia();
  } else {
    nivel = 1;
    mostrarSequencia();
  }
}
