Código C++ (Arduino) 
#include <TM1637Display.h>

const int CLK = 2;
const int DIO = 3;
const int PIN_BOTAO = 4;
const int PIN_BUZZER = 5;

TM1637Display display(CLK, DIO);

unsigned long tempoAnterior = 0;
unsigned long tempoDecorrido = 0;
bool cronometroAtivo = false;

bool botaoPressionadoAnteriormente = false;
unsigned long tempoPressionado = 0;
unsigned long tempoUltimoClique = 0;

const int debounceDelay = 50;      
const int tempoCliqueLongo = 1500;  

void setup() {
  display.setBrightness(0x0f);
  pinMode(PIN_BOTAO, INPUT_PULLUP);
  pinMode(PIN_BUZZER, OUTPUT);
  exibirTempo(0);
}

void loop() {
  int estadoBotao = digitalRead(PIN_BOTAO);
  unsigned long tempoAtual = millis();

  if (estadoBotao == LOW && !botaoPressionadoAnteriormente) {
    tempoPressionado = tempoAtual;
    botaoPressionadoAnteriormente = true;
  }

  if (estadoBotao == LOW && botaoPressionadoAnteriormente) {
    if (tempoAtual - tempoPressionado >= tempoCliqueLongo) {
      tone(PIN_BUZZER, 300, 150);
      delay(150);
      tone(PIN_BUZZER, 150, 250);

      cronometroAtivo = false;
      tempoDecorrido = 0;
      exibirTempo(0);
     
      display.setBrightness(0x00);
      delay(100);
      display.setBrightness(0x0f);
      exibirTempo(0);

      while(digitalRead(PIN_BOTAO) == LOW) { delay(10); }
      botaoPressionadoAnteriormente = false;
    }
  }

  if (estadoBotao == HIGH && botaoPressionadoAnteriormente) {
    unsigned long duracaoClique = tempoAtual - tempoPressionado;

    if (duracaoClique >= debounceDelay && duracaoClique < tempoCliqueLongo) {
      cronometroAtivo = !cronometroAtivo;
     
      if (cronometroAtivo) {
        tone(PIN_BUZZER, 2000, 80);
        delay(100);
        tone(PIN_BUZZER, 2500, 80);
       
        tempoAnterior = millis() - tempoDecorrido;
      } else {
        tone(PIN_BUZZER, 1000, 200);
      }
    }
    botaoPressionadoAnteriormente = false;
  }

  if (cronometroAtivo) {
    tempoDecorrido = millis() - tempoAnterior;
    exibirTempo(tempoDecorrido);
  }
}

void exibirTempo(unsigned long ms) {
  unsigned long totalCenteseconds = ms / 10;
  int segundos = (totalCenteseconds / 100) % 100;
  int centesimos = totalCenteseconds % 100;
  int valorDisplay = (segundos * 100) + centesimos;

  display.showNumberDecEx(valorDisplay, 0b01000000, true);
}

Esquema de Pinagem

  • Sensor de Som Digital Pino 2
  • LEDs 1 a 7 Pinos 3 a 9
  • Sinal do Servo Motor Pino 10
Código C++ (Arduino) 
#include <Servo.h>

const int pinoSom = 2;          
const int pinosLeds[7] = {3, 4, 5, 6, 7, 8, 9}; 
const int pinoServo = 10;       

Servo meuServo;

volatile int contadorBatidas = 0;
volatile unsigned long tempoUltimaBatida = 0;
volatile unsigned long tempoPrimeiraBatida = 0;
volatile bool novaBatidaDetectada = false; 

const unsigned long janelaTempo = 1500; 
const unsigned long debounceTime = 180; 

void setup() {
  Serial.begin(9600);
 
  for (int i = 0; i < 7; i++) {
    pinMode(pinosLeds[i], OUTPUT);
    digitalWrite(pinosLeds[i], LOW); 
  }
 
  pinMode(pinoSom, INPUT);
  attachInterrupt(digitalPinToInterrupt(pinoSom), somDetectado, FALLING);
 
  meuServo.attach(pinoServo);
  meuServo.write(0); 
 
  Serial.println("=========================================");
  Serial.println("Sistema Pronto! Aguardando batidas...");
  Serial.println("=========================================");
}

void loop() {
  if (novaBatidaDetectada) {
    novaBatidaDetectada = false;
    Serial.print("Batida detectada! Contagem atual: ");
    Serial.println(contadorBatidas);
  }

  if (contadorBatidas > 0 && (millis() - tempoPrimeiraBatida > janelaTempo)) {
    detachInterrupt(digitalPinToInterrupt(pinoSom));
   
    int batidasFinais = contadorBatidas;
    if (batidasFinais > 7) batidasFinais = 7; 
   
    Serial.println("\n--- FIM DA JANELA DE TEMPO ---");
    Serial.print("Total final de batidas: ");
    Serial.println(contadorBatidas);

    for (int i = 0; i < 7; i++) {
      digitalWrite(pinosLeds[i], LOW);
    }

    int angulo = batidasFinais * 20;
    meuServo.write(angulo);
   
    int indiceLed = batidasFinais - 1;
    for (int p = 0; p < 3; p++) {
      digitalWrite(pinosLeds[indiceLed], HIGH);
      delay(200);
      digitalWrite(pinosLeds[indiceLed], LOW);
      delay(200);
    }
   
    digitalWrite(pinosLeds[indiceLed], HIGH);
    contadorBatidas = 0;
    attachInterrupt(digitalPinToInterrupt(pinoSom), somDetectado, FALLING);
  }
}

void somDetectado() {
  unsigned long tempoAtual = millis();
 
  if (tempoAtual - tempoUltimaBatida > debounceTime) {
    if (contadorBatidas == 0) {
      tempoPrimeiraBatida = tempoAtual; 
    }
    contadorBatidas++;
    tempoUltimaBatida = tempoAtual;
    novaBatidaDetectada = true; 
  }
}
